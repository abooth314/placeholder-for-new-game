cmake_minimum_required(VERSION 3.25)
project(CurvedWorldPrototype VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CURVED_BUILD_CLIENT "Build native SDL/OpenGL client" ON)
include(cmake/Dependencies.cmake)
add_library(curved_domain STATIC src/math/hyperbolic.cpp src/world/terrain.cpp src/world/spatial.cpp
  src/sim/values.cpp src/sim/entities.cpp src/sim/world.cpp src/sim/tactics.cpp src/sim/combat.cpp
  src/content/catalog.cpp src/content/lua_policy.cpp)
target_include_directories(curved_domain PUBLIC src)
target_link_libraries(curved_domain PRIVATE nlohmann_json::nlohmann_json curved_lua)
target_compile_options(curved_domain PRIVATE $<$<CXX_COMPILER_ID:Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
if(CURVED_BUILD_CLIENT)
  add_library(curved_presentation STATIC src/assets/mesh.cpp src/view/camera.cpp src/view/gl_api.cpp
    src/view/renderer.cpp src/audio/synth.cpp src/audio/audio.cpp src/app/input.cpp src/app/settings.cpp src/ui/console.cpp src/ui/ui.cpp)
  target_include_directories(curved_presentation PUBLIC src)
  target_link_libraries(curved_presentation PUBLIC curved_domain curved_imgui PRIVATE nlohmann_json::nlohmann_json)
  add_executable(curved_world WIN32 src/app/main.cpp)
  target_link_libraries(curved_world PRIVATE curved_presentation nlohmann_json::nlohmann_json)
  if(WIN32)
    target_link_libraries(curved_world PRIVATE opengl32 psapi)
  endif()
  install(TARGETS curved_world RUNTIME DESTINATION .)
endif()
enable_testing()
if(CURVED_BUILD_CLIENT)
  add_executable(curved_presentation_tests tests/presentation_tests.cpp)
  target_link_libraries(curved_presentation_tests PRIVATE curved_presentation)
  foreach(test camera recipes input_settings console)
    add_test(NAME ${test} COMMAND curved_presentation_tests ${test} "${CMAKE_SOURCE_DIR}")
    set_tests_properties(${test} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
add_executable(curved_tests tests/domain_tests.cpp)
target_link_libraries(curved_tests PRIVATE curved_domain nlohmann_json::nlohmann_json)
foreach(test geometry terrain values content lua identity spatial combat diplomacy tactics possession replay bridge_world budgets)
  add_test(NAME ${test} COMMAND curved_tests ${test} "${CMAKE_SOURCE_DIR}")
  set_tests_properties(${test} PROPERTIES TIMEOUT 90)
endforeach()
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs PATTERN references EXCLUDE)
install(DIRECTORY notices/ DESTINATION notices)
install(DIRECTORY assets/ DESTINATION assets)
install(CODE "
  if(NOT EXISTS \"\${CMAKE_INSTALL_PREFIX}/user/enabled-mods.json\")
    file(MAKE_DIRECTORY \"\${CMAKE_INSTALL_PREFIX}/user\")
    file(COPY \"${CMAKE_SOURCE_DIR}/user/enabled-mods.json\" DESTINATION \"\${CMAKE_INSTALL_PREFIX}/user\")
  endif()
  if(NOT EXISTS \"\${CMAKE_INSTALL_PREFIX}/user/mods/wardens\")
    file(MAKE_DIRECTORY \"\${CMAKE_INSTALL_PREFIX}/user/mods\")
    file(COPY \"${CMAKE_SOURCE_DIR}/user/mods/wardens\" DESTINATION \"\${CMAKE_INSTALL_PREFIX}/user/mods\")
  endif()
")
